def load_biologic_mpt(filepath):
    # Find the header row
    skip_rows = 0
    with open(filepath, 'r', encoding='latin-1') as f:
        for i, line in enumerate(f):
            if line.startswith('mode'):
                skip_rows = i
                break
                
    # Read the data
    df = pd.read_csv(filepath, sep='\t', skiprows=skip_rows, encoding='latin-1')
    
    if 'z cycle' in df.columns:
        segregation_col = 'z cycle'
    elif 'cycle number' in df.columns and df['cycle number'].nunique() > 1:
        segregation_col = 'cycle number'
    elif 'loop number' in df.columns:
        segregation_col = 'loop number'
    else:
        segregation_col = None
    
    eis_measurements = {}
    
    if segregation_col and segregation_col in df.columns:
        for cycle_id, group in df.groupby(segregation_col):
            # Bulletproof filter: drop empty rows, 0Hz freq, and 0 Ohm impedance points
            group = group.dropna(subset=['freq/Hz'])
            group = group[(group['freq/Hz'] != 0.0) & (group['Re(Z)/Ohm'] != 0.0)] 
            group = group[group['freq/Hz'] >= 0.09]


            if len(group) == 0:
                continue
                
            freq = group['freq/Hz'].values
            Z_real = group['Re(Z)/Ohm'].values
            Z_imag = group['-Im(Z)/Ohm'].values
            Z = Z_real - 1j * Z_imag
            eis_measurements[int(cycle_id)] = (freq, Z)
    else:
        # Fallback if no cycle column exists
        group = df.dropna(subset=['freq/Hz'])
        group = group[(group['freq/Hz'] != 0.0) & (group['Re(Z)/Ohm'] != 0.0)]
        
        freq = group['freq/Hz'].values
        Z_real = group['Re(Z)/Ohm'].values
        Z_imag = group['-Im(Z)/Ohm'].values
        Z = Z_real - 1j * Z_imag
        eis_measurements[1] = (freq, Z)
        
    return eis_measurements

import numpy as np
import ipywidgets as widgets
import matplotlib.pyplot as plt
from scipy.optimize import nnls

def compute_drt(freq, Z_imag, lam):
    """
    Computes Distribution of Relaxation Times using Tikhonov Regularization (NNLS).
    """
    omega = 2 * np.pi * freq
    tau = 1 / omega  # Time constants corresponding to measured frequencies
    
    # 1. Build the design matrix A (for imaginary impedance part)
    A = np.zeros((len(omega), len(tau)))
    for i, w in enumerate(omega):
        for j, t in enumerate(tau):
            A[i, j] = (w * t) / (1 + (w * t)**2)
            
    # 2. Build the regularization matrix L (2nd derivative penalty for smoothness)
    n = len(tau)
    L = np.zeros((n-2, n))
    for i in range(n-2):
        L[i, i] = 1
        L[i, i+1] = -2
        L[i, i+2] = 1
        
    # 3. Augment matrices for Tikhonov regularization in NNLS
    A_aug = np.vstack([A, np.sqrt(lam) * L])
    b_aug = np.hstack([Z_imag, np.zeros(n-2)])
    
    # 4. Solve for gamma (must be strictly non-negative)
    gamma_opt, _ = nnls(A_aug, b_aug)
    
    return tau, gamma_opt

# --- Interactive DRT Plotting ---
def update_drt(cycle_id, log_lam):
    lam = 10**log_lam
    
    # Get the data for the selected cycle
    freq, Z = eis_dict[cycle_id]
    Z_imag_data = -Z.imag  # We fit against the negative imaginary part
    
    # Compute the DRT!
    tau, gamma = compute_drt(freq, Z_imag_data, lam)
    
    # Plot it

    ax_drt.grid(True, alpha=0.5)

import numpy as np
import ipywidgets as widgets
import matplotlib.pyplot as plt
import matplotlib.cm as cm
from scipy.interpolate import interp1d
from IPython.display import display

# 1. Set a fixed Regularization Parameter
lam = 10**-2.0 
all_cycle_ids = sorted(list(eis_dict.keys()))

# 2. Pre-compute the common X-axis (tau) based on the last cycle
master_freq, _ = eis_dict[all_cycle_ids[-1]]
tau_ref = 1 / (2 * np.pi * master_freq)

# 3. PRE-COMPUTE all DRT curves
gamma_dict = {}
for c in all_cycle_ids:
    freq, Z = eis_dict[c]
    tau, gamma = compute_drt(freq, -Z.imag, lam)
    f_interp = interp1d(np.log10(tau), gamma, bounds_error=False, fill_value=0.0)
    gamma_dict[c] = f_interp(np.log10(tau_ref))

# Find the absolute maximum intensity across all cycles to set our slider scale
global_gamma_max = max([np.max(g) for g in gamma_dict.values()])

# 4. Interactive Plotting Function
def update_drt_multi(cycle_range, intensity_max, log_tau_max):
    start, end = cycle_range
    selected_cycles = [c for c in all_cycle_ids if start <= c <= end]
    
    if len(selected_cycles) < 2:
        print("Please select a range of at least 2 cycles to generate the Contour Map.")
        return
        
    gamma_matrix = np.array([gamma_dict[c] for c in selected_cycles])
    
    # Calculate tau max from the logarithmic slider
    tau_max = 10**log_tau_max
    tau_min = np.min(tau_ref) # Keep the left side fixed to the minimum tau
    
    # Create the figure
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(15, 6))

    # --- PLOT 1: 2D Overlapping Curves ---
    for i, c in enumerate(selected_cycles):
        color = cm.plasma(c / max(all_cycle_ids)) if max(all_cycle_ids) > 0 else 'blue'
        ax1.semilogx(tau_ref, gamma_matrix[i], color=color, linewidth=1.5)

    # Set the dynamic limits!
    ax1.set_xlim(left=tau_min, right=tau_max)
    ax1.set_ylim(bottom=0, top=intensity_max)
    
    ax1.set_xlabel('Relaxation Time $\\tau$ (s)', fontsize=12)
    ax1.set_ylabel('$\gamma(\\tau)$ ($\Omega$)', fontsize=12)
    ax1.set_title(f'Overlapping DRT (Cycles {start}-{end})', fontsize=14)
    ax1.grid(True, alpha=0.5)

    # Colorbar
    sm = plt.cm.ScalarMappable(cmap=cm.plasma, norm=plt.Normalize(vmin=min(all_cycle_ids), vmax=max(all_cycle_ids)))
    cbar1 = fig.colorbar(sm, ax=ax1)
    cbar1.set_label('Cycle Number')

    # --- PLOT 2: 2D Contour Map (Heatmap) ---
    X, Y = np.meshgrid(tau_ref, selected_cycles)
    
    # We use np.linspace to force the contour levels to map from 0 up to intensity_max.
    # extend='max' means anything higher than intensity_max is just colored with the top color (yellow)
    contour = ax2.contourf(X, Y, gamma_matrix, levels=np.linspace(0, intensity_max, 30), cmap='viridis', extend='max')

    ax2.set_xscale('log') 
    ax2.set_xlim(left=tau_min, right=tau_max) # Apply X-axis crop to contour map too
    ax2.set_xlabel('Relaxation Time $\\tau$ (s)', fontsize=12)
    ax2.set_ylabel('Cycle Number', fontsize=12)
    ax2.set_title(f'DRT Contour Map (Cycles {start}-{end})', fontsize=14)

    cbar2 = fig.colorbar(contour, ax=ax2)
    cbar2.set_label('$\gamma(\\tau)$ Intensity ($\Omega$)')

    plt.tight_layout()
    plt.show()

# 5. Create Sliders
cycle_range_slider = widgets.IntRangeSlider(
    value=[all_cycle_ids[0], all_cycle_ids[-1]],
    min=min(all_cycle_ids), max=max(all_cycle_ids), step=1,
    description='Cycles:', continuous_update=False, orientation='horizontal'
)

intensity_slider = widgets.FloatSlider(
    value=global_gamma_max, min=0.1, max=global_gamma_max*1.2, step=0.5,
    description='Intensity:', continuous_update=False, orientation='horizontal',
    tooltip="Lower this to artificially boost the brightness of tiny peaks."
)

tau_max_slider = widgets.FloatSlider(
    value=np.log10(np.max(tau_ref)), min=np.log10(np.min(tau_ref)), max=np.log10(np.max(tau_ref)), step=0.1,
    description='log(Max $\\tau$):', continuous_update=False, orientation='horizontal',
    tooltip="Slide left to zoom in and hide the high-tau (low frequency) noise."
)

# Link all sliders!
widgets.interact(update_drt_multi, cycle_range=cycle_range_slider, intensity_max=intensity_slider, log_tau_max=tau_max_slider);
